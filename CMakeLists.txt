cmake_minimum_required(VERSION 3.16)
project(avbd_demo3d VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(avbd3d src/avbd3d.cpp)
target_include_directories(avbd3d PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
  target_compile_options(avbd3d PRIVATE /W4)
else()
  target_compile_options(avbd3d PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(avbd3d_tests tests/tests.cpp)
target_link_libraries(avbd3d_tests PRIVATE avbd3d)
add_executable(avbd3d_demo src/demo.cpp)
target_link_libraries(avbd3d_demo PRIVATE avbd3d)
add_executable(avbd3d_stress src/stress.cpp)
target_link_libraries(avbd3d_stress PRIVATE avbd3d)
add_executable(avbd3d_sky_capture src/sky_capture.cpp)
target_link_libraries(avbd3d_sky_capture PRIVATE avbd3d)
add_executable(avbd3d_limitations tests/known_limitations.cpp)
target_link_libraries(avbd3d_limitations PRIVATE avbd3d)

enable_testing()
add_test(NAME physics_suite COMMAND avbd3d_tests)
add_executable(avbd3d_advancement tests/advancement.cpp)
target_link_libraries(avbd3d_advancement PRIVATE avbd3d)
add_test(NAME advancement COMMAND avbd3d_advancement)
add_executable(avbd3d_stages src/stages.cpp)
target_link_libraries(avbd3d_stages PRIVATE avbd3d)
add_executable(avbd3d_benchmark src/benchmark.cpp)
target_link_libraries(avbd3d_benchmark PRIVATE avbd3d)

add_executable(avbd3d_ccd_parallel_tests tests/ccd_parallel.cpp)
target_link_libraries(avbd3d_ccd_parallel_tests PRIVATE avbd3d)
add_test(NAME ccd_parallel COMMAND avbd3d_ccd_parallel_tests)

option(AVBD_ENABLE_OPENMP "Enable parallel graph-color solver with OpenMP" ON)
if(AVBD_ENABLE_OPENMP)
  find_package(OpenMP QUIET COMPONENTS CXX)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(avbd3d PUBLIC OpenMP::OpenMP_CXX)
    target_compile_definitions(avbd3d PRIVATE AVBD_HAS_OPENMP=1)
  endif()
endif()
add_executable(avbd3d_parallel_benchmark src/benchmark_parallel.cpp)
target_link_libraries(avbd3d_parallel_benchmark PRIVATE avbd3d)
add_executable(avbd3d_milestones src/milestones.cpp)
target_link_libraries(avbd3d_milestones PRIVATE avbd3d)

add_executable(avbd3d_rotational_ccd_tests tests/rotational_ccd.cpp)
target_link_libraries(avbd3d_rotational_ccd_tests PRIVATE avbd3d)
add_test(NAME rotational_ccd COMMAND avbd3d_rotational_ccd_tests)
add_executable(avbd3d_rotational_capture src/rotational_capture.cpp)
target_link_libraries(avbd3d_rotational_capture PRIVATE avbd3d)

add_executable(avbd3d_scaling_tests tests/scaling.cpp)
target_link_libraries(avbd3d_scaling_tests PRIVATE avbd3d)
add_test(NAME scaling COMMAND avbd3d_scaling_tests)
add_executable(avbd3d_scaling_benchmark src/benchmark_scaling.cpp)
target_link_libraries(avbd3d_scaling_benchmark PRIVATE avbd3d)
add_executable(avbd3d_scaling_capture src/scaling_capture.cpp)
target_link_libraries(avbd3d_scaling_capture PRIVATE avbd3d)
